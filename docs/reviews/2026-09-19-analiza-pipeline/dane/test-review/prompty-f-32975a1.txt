 * gotowy „stempel" punktu, więc musi wyprzedzić `draw`.
 */
declare module 'simpleheat' {
  /** Punkt w układzie współrzędnych canvasu: `[x, y, waga]`. */
  export type SimpleheatPoint = readonly [number, number, number];

  /** Mapa `próg (0–1) → kolor CSS` — gradient rysowanej nakładki. */
  export type SimpleheatGradient = Readonly<Record<number, string>>;

  export interface SimpleheatInstance {
    data(points: readonly SimpleheatPoint[]): SimpleheatInstance;
    max(max: number): SimpleheatInstance;
    add(point: SimpleheatPoint): SimpleheatInstance;
    clear(): SimpleheatInstance;
    radius(radius: number, blur?: number): SimpleheatInstance;
    gradient(gradient: SimpleheatGradient): SimpleheatInstance;
    resize(): SimpleheatInstance;
    draw(minOpacity?: number): SimpleheatInstance;
  }

  export default function simpleheat(canvas: HTMLCanvasElement): SimpleheatInstance;
}
